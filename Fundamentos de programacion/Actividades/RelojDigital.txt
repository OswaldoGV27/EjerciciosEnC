		Mientras Minuto <= 59 Hacer
			Mientras Segundo0 <= 59 Hacer

				FinSi

				SiNo

				FinSi
				
				Si Segundo0 > 9 Entonces
					Escribir ":", Segundo0;
				SiNo
					Escribir ":0", Segundo0;
				FinSi
				Segundo0 = Segundo0 +1;
				Esperar 1000 Milisegundos;
			FinMientras
			Segundo0 = 0
			Minuto = Minuto +1
		FinMientras
		Minuto = 0;
		Hora = Hora +1;
